// ============================================================
// SCREENER: Trampa Alcista v2.0
// Low Float NASDAQ/NYSE — Daily (uso habitual: premercado)
// Objetivo: detectar post-pump en fase de distribución temprana → candidatos SHORT
// Edge: Koncorde detecta la salida institucional mientras el retail sigue comprando
//       Vigía10 confirma el giro de momentum — Ichimoku marca el deterioro estructural
//
// FLUJO OPERATIVO OBLIGATORIO:
//   1. Screener en premercado → watchlist de candidatos con distribución confirmada
//   2. Validar en Finviz: float real, país, market cap, performance semanal
//   3. Validar manualmente: ¿reverse split reciente? ¿offering SEC? ¿empresa basura?
//   4. Verificar borrow disponible y coste en broker ANTES de cualquier posición
//   5. La entrada real es intraday — esperar agotamiento + pérdida VWAP en apertura
//   REGLA DE ORO: NUNCA shortear el pump vertical — el screener filtra los más activos
//
// CAMBIOS VS v1.0:
//   [FIX] noEnPump: excluye stocks subiendo >20% hoy — pump todavía activo en premercado
//     Razón: AGAPE ATP (+62%), QUANTUM (+19%), AUDDIA (+20%) no son setups — son el pump
//   [FIX] hayDistribucion más estricto: azulAceleraBaj solo cuenta si azul < 10
//     Razón: GCT SEMIC (azul=8.63) y NEO-CONC (azul=13.29) eran falsos positivos
//     Un azul cayendo de 13 a 11 no es distribución — es enfriamiento de acumulación
//
// AJUSTE DE SEÑALES:
//   Pocas señales → bajar scoreMin a 40 o pumpMinPct a 20
//   Muchas señales → subir scoreMin a 70 o pumpMinPct a 40
//   noEnPump demasiado agresivo → subir umbral a 30
// ============================================================

// ---- PARÁMETROS AJUSTABLES ----
minPrice    = 1
maxPrice    = 20
minDolarVol = 1000000   // liquidez mínima — micro stocks (volumen premercado es bajo: filtrar a mano)
pumpMinPct  = 30        // pump mínimo en últimos 5 días (%)
scoreMin    = 55        // umbral mínimo de calidad combinada

// ================================================================
// GATE BASE — precio, liquidez, volatilidad
// ================================================================
volMedia20  = Average[20](Volume)
dolarVol    = close * volMedia20
atr14       = AverageTrueRange[14]
atrRelativo = atr14 / (close + 0.001)

baseOk = close >= minPrice AND close <= maxPrice AND dolarVol >= minDolarVol AND atrRelativo > 0.05

// ================================================================
// ANTI-PUMP ACTIVO — [FIX v2.0]
// Excluye stocks que siguen subiendo hoy (pump todavía en marcha)
// En premercado detecta los que ya llevan >15% de subida antes de abrir
// Estos son candidatos para MAÑANA, no para hoy
// ================================================================
movHoy    = (close - close[1]) / (close[1] + 0.001) * 100
noEnPump  = movHoy <= 15

// ================================================================
// DETECCIÓN DE PUMP RECIENTE
// ================================================================

// Máximo alcanzado en los últimos 5 días (incluyendo hoy)
maxPump5 = Highest[5](High)

// Pump %: cuánto subió desde el cierre de hace 5 días hasta el máximo reciente
pumpPct    = (maxPump5 - close[5]) / (close[5] + 0.001) * 100
pumpFuerte = pumpPct >= pumpMinPct

// Precio aún por encima del nivel pre-pump (pump no completamente revertido)
extendido = close > close[5]

// Distancia desde el pico del pump — cuánto ha caído ya desde el máximo
distDeMaximo = (maxPump5 - close) / (maxPump5 + 0.001) * 100
noTardio     = distDeMaximo < 50   // excluye si ya cayó más del 50% desde el pico

// Volumen del pump: máximo volumen en los últimos 5 días vs media 20
// Usa Highest[5](Volume) para capturar el día del pump aunque hoy Volume=0 en premercado
volPumpMax = Highest[5](Volume) / (volMedia20 + 0.001)

// ================================================================
// BLOQUE KONCORDE — señales de distribución institucional
// NVI sube en días de poco volumen = smart money / institucionales
// Azul cayendo mientras precio sube = están saliendo en silencio
// ================================================================
nvi    = NegativeVolumeIndex(close)
nvim   = ExponentialAverage[3](nvi)
nvimax = Highest[90](nvim)
nvimin = Lowest[90](nvim)
azul   = (nvi - nvim) * 100 / Max(nvimax - nvimin, 0.001)

// Distribución silenciosa: precio sube pero azul ya cae — señal pre-cruce más valiosa
divergenciaBaj = azul < azul[3] AND close > close[3]
// Salida activa: azul cayendo 3 sesiones — solo válido si azul < 10 [FIX v2.0]
// Si azul es muy positivo (>10), caer no es distribución sino enfriamiento de acumulación
azulAceleraBaj = azul < azul[1] AND azul[1] < azul[2] AND azul < 10
// Institucionales ya fuera
azulBajoCero   = azul < 0

// Puerta Koncorde: al menos una señal real de distribución activa
hayDistribucion = divergenciaBaj OR azulAceleraBaj OR azulBajoCero

// ================================================================
// BLOQUE VIGÍA10 — giro de momentum
// ================================================================
ignored, ignored, ignored, ignored, ignored, SV, MR, ignored = CALL "Blai5 VIGIA 10"
filtro = ExponentialAverage[20](SV)

svBajando    = SV < SV[1] AND SV[1] < SV[2]
crossSV0neg  = SV CROSSES UNDER 0
bsSV0neg     = barssince(crossSV0neg)
crossSVMRneg = SV CROSSES UNDER MR
bsSVMRneg    = barssince(crossSVMRneg)

// ================================================================
// BLOQUE ICHIMOKU — deterioro estructural
// En stocks post-pump el precio está sobre la nube;
// la señal relevante es la pérdida de Tenkan y Kijun
// ================================================================
tenkan     = (Highest[9](High) + Lowest[9](Low)) / 2
kijun      = (Highest[26](High) + Lowest[26](Low)) / 2
ssa0       = (tenkan + kijun) / 2
ssb0       = (Highest[52](High) + Lowest[52](Low)) / 2
ssa        = ssa0[26]
ssb        = ssb0[26]
kumoBottom = Min(ssa, ssb)

// ================================================================
// SCORING KONCORDE — distribución institucional
// ================================================================

// [K0] 35 pts — SEÑAL PREMIUM: distribución silenciosa pre-cruce
// Institucionales saliendo mientras precio sube → la señal más valiosa y temprana
sK0 = divergenciaBaj * 35

// [K1] 20 pts — Azul cayendo 3 sesiones seguidas con azul < 10 (salida sostenida y activa)
sK1 = azulAceleraBaj * 20

// [K2] 15 pts — Azul bajo cero (institucionales ya fuera del stock)
sK2 = azulBajoCero * 15

// [K3] 10 pts — Azul bajo cero Y acelerando (máxima convicción de salida)
sK3 = (azulBajoCero AND azulAceleraBaj) * 10

scoreK = sK0 + sK1 + sK2 + sK3   // máx teórico: 80

// ================================================================
// SCORING VIGÍA10 — flip de momentum bajista
// ================================================================

// [V1] 30 pts — SV cruza 0 hacia abajo (señal bajista máxima de Vigía10)
sV1 = (bsSV0neg <= 3 AND SV < 0) * 30

// [V2] 20 pts — SV cruza bajo MR (señal temprana antes del cruce de cero)
sV2 = (bsSVMRneg <= 5) * 20

// [V3] 15 pts — SV negativo y cayendo (momentum bajista confirmado y sostenido)
sV3 = (SV < 0 AND svBajando) * 15

// [V4] 10 pts — Filtro girando a la baja (inercia lenta confirma dirección bajista)
sV4 = (filtro < filtro[1] AND filtro < filtro[3]) * 10

// [V5] 8 pts — SV < MR y filtro negativo (alineación bajista completa)
sV5 = (SV < MR AND filtro < 0) * 8

scoreV = sV1 + sV2 + sV3 + sV4 + sV5   // máx teórico: 83

// ================================================================
// SCORING ICHIMOKU — deterioro estructural de precio
// ================================================================

// [I1] 20 pts — Precio bajo la Kijun (pérdida del equilibrio de 26 sesiones)
sI1 = (close < kijun) * 20

// [I2] 10 pts — Precio bajo la Tenkan pero sobre Kijun (primer aviso)
sI2 = (close < tenkan AND close >= kijun) * 10

// [I3] 8 pts — Nube bajista futura (SSB > SSA: proyección negativa a 26 sesiones)
sI3 = (ssb > ssa) * 8

// [I4] 5 pts — Precio bajo la nube (estructura bajista total — tardío en pumps)
sI4 = (close < kumoBottom) * 5

scoreI = sI1 + sI2 + sI3 + sI4   // máx teórico: 43

// ================================================================
// SCORING PUMP — calidad y extensión del pump detectado
// Cuanto más extremo el pump, más potencial de colapso
// ================================================================

// [P1] 15 pts — Pump muy violento (>= 60% en 5 días)
sP1 = (pumpPct >= 60) * 15

// [P2] 8 pts — Pump moderado (30–60%)
sP2 = (pumpPct >= pumpMinPct AND pumpPct < 60) * 8

// [P3] 12 pts — Precio extremadamente extendido sobre su media de 20 días
sP3 = (close > Average[20](close) * 1.5) * 12

// [P4] 8 pts — Moderadamente extendido sobre media
sP4 = (close > Average[20](close) * 1.2 AND close <= Average[20](close) * 1.5) * 8

// [P5] 12 pts — Volatilidad extrema (ATR relativo > 10% del precio)
sP5 = (atrRelativo > 0.10) * 12

// [P6] 6 pts — Volatilidad alta (ATR 5–10% del precio)
sP6 = (atrRelativo > 0.05 AND atrRelativo <= 0.10) * 6

// [P7] 10 pts — Inicio de caída desde el techo (10–30% desde el máximo del pump)
sP7 = (distDeMaximo >= 10 AND distDeMaximo < 30) * 10

// [P8] 15 pts — Pump con volumen explosivo (>= 5x el volumen medio de 20 días)
sP8 = (volPumpMax >= 5) * 15

// [P9] 8 pts — Pump con volumen elevado (2x–5x la media)
sP9 = (volPumpMax >= 2 AND volPumpMax < 5) * 8

scoreP = sP1 + sP2 + sP3 + sP4 + sP5 + sP6 + sP7 + sP8 + sP9   // máx teórico: 94

// ================================================================
// SCORE FINAL Y CONDICIÓN
// ================================================================
score  = scoreK + scoreV + scoreI + scoreP   // máx teórico: 300

// PUERTAS NÚCLEO:
//   baseOk:          precio, liquidez y volatilidad mínima
//   noEnPump:        excluye stocks todavía subiendo >20% hoy [NUEVO v2.0]
//   pumpFuerte:      hubo un pump real de al menos pumpMinPct%
//   extendido:       precio aún sobre el nivel pre-pump (no ya revertido)
//   noTardio:        no cayó más del 50% desde el pico (entrada no tardía)
//   hayDistribucion: Koncorde confirma salida institucional real [FIX v2.0]
condicion = baseOk AND noEnPump AND pumpFuerte AND extendido AND noTardio AND hayDistribucion AND score >= scoreMin

SCREENER[score * condicion](score AS "Score", pumpPct AS "Pump%", distDeMaximo AS "DistMax%", movHoy AS "Hoy%", divergenciaBaj AS "DivBaj", azul AS "Azul", SV AS "SV", scoreK AS "ScoreK", scoreV AS "ScoreV", close AS "Precio", dolarVol AS "DolarVol")
