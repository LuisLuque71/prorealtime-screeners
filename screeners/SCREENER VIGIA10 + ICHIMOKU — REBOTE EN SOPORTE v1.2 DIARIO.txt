// ============================================================
// SCREENER VIGIA10 + ICHIMOKU — REBOTE EN SOPORTE v1.2 DIARIO
// Tendencia previa → Corrección → Toque soporte → Rebote
// Ventana toque: 2-8 sesiones · Objetivo: +15-20%
// Timeframe: DIARIO · Alta convicción · Pocas señales
// ============================================================
// Si salen 0 resultados: baja tramoMin a 0.15 o scoreMin a 110
// ============================================================

// ---- PARÁMETROS ----
minVolDolar    = 500000
minPrice       = 8
maxPrice       = 500
tolerSoporte   = 0.040
kijunSlopeBars = 10
minTendPrevia  = 0.10
scoreMin       = 130
tramoMin       = 0.20

// ---- DATOS BÁSICOS ----
c = Close
v = Volume
volDolarHoy = v * c
volDolarMed = Average[20](v * c)
cLiquidez   = volDolarMed >= minVolDolar AND volDolarHoy >= minVolDolar * 0.4
cPrecio     = c >= minPrice AND c <= maxPrice

// ---- VIGÍA10 ----
ignored, vpm, astro, ignored, ignored, filtro, sv, mr = CALL "Blai5 VIGIA 10"
avgVpm = Average[20](vpm)
filtroPosNow      = filtro > 0
filtroGirando     = filtro > filtro[2]
svPos             = sv > 0
astroPos          = astro > 0
vpmOk             = vpm >= avgVpm
svSobreMrOGirando = sv > mr OR (sv > sv[1] AND sv > sv[2])

// ---- ICHIMOKU ----
tenkan = (Highest[9](High)  + Lowest[9](Low))  / 2
kijun  = (Highest[26](High) + Lowest[26](Low)) / 2
ssa0   = (tenkan + kijun) / 2
ssb0   = (Highest[52](High) + Lowest[52](Low)) / 2
ssa    = ssa0[26]
ssb    = ssb0[26]

kumoBot     = Min(ssa, ssb)
kumoAlcista = ssa > ssb

// ---- EMA200 ----
ema200 = Average[200](Close)

// ---- TENDENCIA PREVIA ----
max60           = Highest[60](High)
tendenciaPrevia = max60 > c * (1 + minTendPrevia)

// ---- DETECCIÓN TOQUE SOPORTE (ventana 2-8 sesiones) ----
tK2 = abs(Close[2] - kijun[2]) / (kijun[2] + 0.0001) < tolerSoporte
tK3 = abs(Close[3] - kijun[3]) / (kijun[3] + 0.0001) < tolerSoporte
tK4 = abs(Close[4] - kijun[4]) / (kijun[4] + 0.0001) < tolerSoporte
tK5 = abs(Close[5] - kijun[5]) / (kijun[5] + 0.0001) < tolerSoporte
tK6 = abs(Close[6] - kijun[6]) / (kijun[6] + 0.0001) < tolerSoporte
tK7 = abs(Close[7] - kijun[7]) / (kijun[7] + 0.0001) < tolerSoporte
tK8 = abs(Close[8] - kijun[8]) / (kijun[8] + 0.0001) < tolerSoporte
tocoKijun = tK2 OR tK3 OR tK4 OR tK5 OR tK6 OR tK7 OR tK8

tS2 = abs(Close[2] - ssb[2]) / (ssb[2] + 0.0001) < tolerSoporte
tS3 = abs(Close[3] - ssb[3]) / (ssb[3] + 0.0001) < tolerSoporte
tS4 = abs(Close[4] - ssb[4]) / (ssb[4] + 0.0001) < tolerSoporte
tS5 = abs(Close[5] - ssb[5]) / (ssb[5] + 0.0001) < tolerSoporte
tS6 = abs(Close[6] - ssb[6]) / (ssb[6] + 0.0001) < tolerSoporte
tS7 = abs(Close[7] - ssb[7]) / (ssb[7] + 0.0001) < tolerSoporte
tS8 = abs(Close[8] - ssb[8]) / (ssb[8] + 0.0001) < tolerSoporte
tocoSSB = tS2 OR tS3 OR tS4 OR tS5 OR tS6 OR tS7 OR tS8

tE2 = abs(Close[2] - ema200[2]) / (ema200[2] + 0.0001) < tolerSoporte
tE3 = abs(Close[3] - ema200[3]) / (ema200[3] + 0.0001) < tolerSoporte
tE4 = abs(Close[4] - ema200[4]) / (ema200[4] + 0.0001) < tolerSoporte
tE5 = abs(Close[5] - ema200[5]) / (ema200[5] + 0.0001) < tolerSoporte
tE6 = abs(Close[6] - ema200[6]) / (ema200[6] + 0.0001) < tolerSoporte
tE7 = abs(Close[7] - ema200[7]) / (ema200[7] + 0.0001) < tolerSoporte
tE8 = abs(Close[8] - ema200[8]) / (ema200[8] + 0.0001) < tolerSoporte
tocoEMA200 = tE2 OR tE3 OR tE4 OR tE5 OR tE6 OR tE7 OR tE8

tocoSoportePrioritario = tocoKijun OR tocoSSB
tocoDoble              = tocoKijun AND tocoSSB
tocoTriple             = tocoKijun AND tocoSSB AND tocoEMA200

// ---- CONFIRMACIÓN REBOTE ----
sobreKijun    = c > kijun
sobreSSB      = c > ssb
confirmacion1 = Close[1] > Close[2]
confirmacion2 = Close[1] > Close[2] AND c > Close[1]
slope3        = (c - Close[3]) / (Close[3] + 0.0001)
reboteActivo  = slope3 > 0 AND c > Close[1]

// ---- ICHIMOKU FONDO SANO ----
kijunRising         = kijun > kijun[kijunSlopeBars]
chikouLibre         = c > Close[26]
kumoFuturaAlc       = ssa0 > ssb0
precioEnZonaAlcista = c > kumoBot OR kumoAlcista

// ---- RECORRIDO LIBRE AL MÁXIMO ----
recorridoLibre = (max60 - c) / (c + 0.0001)

// ---- SCORE DIARIO ----
scoreRebote = (tocoSSB * 25) + (tocoKijun * 20) + (tocoEMA200 * 10) + (tocoDoble * 30) + (tocoTriple * 80) + (confirmacion2 * 20) + (confirmacion1 * 10) + (kumoAlcista * 10) + (kijunRising * 12) + (chikouLibre * 8) + (kumoFuturaAlc * 8) + (filtroPosNow * 18) + (filtroGirando * 10) + (svSobreMrOGirando * 12) + (vpmOk * 5) + (astroPos * 5) + (recorridoLibre * 150)

// ---- CONDICIÓN FINAL ----
soporteOk     = tocoSoportePrioritario
reboteOk      = reboteActivo AND (sobreKijun OR sobreSSB)
vigiaOk       = svPos AND filtroGirando
ichimokuFondo = precioEnZonaAlcista
calidadOk     = scoreRebote >= scoreMin AND recorridoLibre >= tramoMin

condicion = cLiquidez AND cPrecio AND tendenciaPrevia AND soporteOk AND reboteOk AND vigiaOk AND ichimokuFondo AND calidadOk

SCREENER[condicion](scoreRebote AS "Score", recorridoLibre*100 AS "Tramo%", tocoSSB AS "TocóSSB", tocoKijun AS "TocóKij", tocoEMA200 AS "TocóEMA", tocoTriple AS "Triple", confirmacion2 AS "2Conf", sv AS "SV", filtro AS "Filtro", Close AS "Precio")
TAKANA EN KUMO v5.5
