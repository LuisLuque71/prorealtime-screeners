// ============================================================
// SCREENER: Vigía10 Pre-Breakout v1.3
// Penny/Micro stocks NYSE/NASDAQ - Daily
// ============================================================

// --- HARD GATES ---
volMedia20 = average[20](volume)
dolarVol   = close * volMedia20
baseOk     = close >= 1 AND close <= 10 AND dolarVol >= 1000000

// --- VIGÍA10 ---
ignored, ignored, ignored, ignored, ignored, ignored, sv, mr = CALL "Blai5 VIGIA 10"
filtro = ExponentialAverage[50](sv)
astro  = sv - mr

// --- CONDICIONES BÁSICAS ---
svSubiendo  = sv > sv[1] AND sv[1] >= sv[3]
svConFuerza = (sv - sv[1]) >= (sv[1] - sv[2])

// --- COMPRESIÓN PREVIA ---
amplitudReciente = highest[5](sv) - lowest[5](sv)
amplitudPrevia   = highest[15](sv[5]) - lowest[15](sv[5])
postSqueeze = amplitudPrevia <= 15 AND amplitudReciente >= amplitudPrevia * 1.2

// --- TIPOS DE SEÑAL ---
tipoB = sv > 0 AND sv[3] <= 0 AND svSubiendo
tipoA = sv < 0 AND sv > -5 AND sv > mr AND svSubiendo AND NOT tipoB
tipoC = sv > 0 AND mr > 0 AND filtro > 0 AND NOT tipoB
tipoD = filtro < 0 AND (filtro - filtro[3]) > (sv - sv[3]) AND filtro > filtro[3] AND sv < 0 AND svSubiendo AND NOT tipoA AND NOT tipoB

haySenal = tipoA OR tipoB OR tipoC OR tipoD

// --- SCORING ---
s1 = tipoB * 40 + tipoA * 25 + tipoC * 15 + tipoD * 10
s2 = (filtro > 0) * 15
s3 = (astro > 0 AND astro[3] <= 0) * 15
s4 = (mr > 0) * 10
s5 = svConFuerza * 10
s6 = (astro > 0) * 8
s7 = (filtro > filtro[3]) * 5
s8 = postSqueeze * 12

score = s1 + s2 + s3 + s4 + s5 + s6 + s7 + s8

// --- TIPO LEGIBLE ---
tipoNum = tipoB * 4 + tipoA * 3 + tipoC * 2 + tipoD * 1

// --- CONDICIÓN FINAL ---
scoreOrden = tipoNum * 1000 + score
condicion  = baseOk AND svSubiendo AND haySenal AND score >= 65

SCREENER[scoreOrden * condicion](tipoNum,score,sv,mr,filtro,close)
