// ============================================================
// SCREENER: Trampa Alcista v3.1
// Low Float NASDAQ/NYSE — Daily (uso habitual: premercado)
// Objetivo: detectar post-pump en fase de distribución temprana → candidatos SHORT
// Edge: serial offenders probados + Koncorde distribución + Vigía10 giro momentum
//
// FLUJO OPERATIVO OBLIGATORIO:
//   1. Screener en premercado → watchlist de candidatos con distribución confirmada
//   2. Validar en Finviz: float real, país, market cap, performance semanal
//   3. Validar manualmente: ¿reverse split reciente? ¿offering SEC? ¿empresa basura?
//   4. Verificar borrow disponible y coste en broker ANTES de cualquier posición
//   5. La entrada real es intraday — esperar agotamiento + pérdida VWAP en apertura
//   REGLA DE ORO: NUNCA shortear el pump vertical — el screener filtra los más activos
//
// JERARQUÍA DE GATES (en orden de importancia):
//   1. hadCrash90  — el más importante: solo serial offenders con crash >90% en 180 días
//   2. hayDistribucion — Koncorde confirma salida institucional activa
//   3. pumpFuerte + extendido + noTardio — detección del ciclo pump actual
//   4. baseOk + noEnPump — filtros de precio/liquidez/timing
//
// NOTA SOBRE KONCORDE EN SERIAL OFFENDERS:
//   NVI solo cambia en días de volumen bajo → los crash days (volumen extremo) NO
//   distorsionan azul. Koncorde es fiable para estos stocks. El crash previo
//   genera acumulación post-crash (azul sube en silencio); la distribución actual
//   hace caer azul — exactamente lo que divergenciaBaj detecta.
//
// CAMBIOS VS v3.0:
//   [NUEVO] sC1/sC2: scoring de frescura del crash
//     Un crash de hace 20 días es mucho más relevante que uno de hace 170.
//     bsCrash ya estaba calculado pero no sumaba — ahora sí.
//     sC1 (20pts): crash ≤ 60 días — patrón fresco, FOMO retail más activo
//     sC2 (10pts): crash 60-180 días — patrón reconocible pero más frío
//   [FIX] noTardio: de 50% a 40%
//     Serial offenders no caen ordenadamente — colapsan verticalmente.
//     40% off peak en un low float chino ya es inicio de dump, no distribución.
//
// AJUSTE DE SEÑALES:
//   Pocas señales → bajar scoreMin a 45, pumpMinPct a 20, o crashLookback a 270
//   Muchas señales → subir scoreMin a 65 o pumpMinPct a 40
//   noEnPump demasiado agresivo → subir umbral a 20
// ============================================================

// ---- PARÁMETROS AJUSTABLES ----
minPrice      = 1
maxPrice      = 20
minDolarVol   = 1000000   // liquidez mínima (volumen premercado es bajo: filtrar a mano)
pumpMinPct    = 30        // pump mínimo en últimos 5 días (%)
scoreMin      = 50        // umbral ajustado — universo más pequeño tras gate crash
crashLookback = 180       // días hacia atrás para buscar el crash histórico

// ================================================================
// GATE BASE — precio, liquidez, volatilidad
// ================================================================
volMedia20  = Average[20](Volume)
dolarVol    = close * volMedia20
atr14       = AverageTrueRange[14]
atrRelativo = atr14 / (close + 0.001)

baseOk = close >= minPrice AND close <= maxPrice AND dolarVol >= minDolarVol AND atrRelativo > 0.05

// ================================================================
// ANTI-PUMP ACTIVO
// Excluye stocks que siguen subiendo hoy (pump todavía en marcha)
// En premercado detecta los que ya llevan >15% de subida antes de abrir
// ================================================================
movHoy   = (close - close[1]) / (close[1] + 0.001) * 100
noEnPump = movHoy <= 15

// ================================================================
// HISTORIAL DE CRASH — GATE PRINCIPAL [desde v3.0]
// Detecta si el stock ya sufrió un colapso >90% en 1-4 días
// en algún momento de los últimos 180 días.
// crashBar: en cada barra histórica, ¿cayó el cierre >90% vs el máximo de los
//           4 cierres anteriores? NVI no cambia en días de alto volumen (crash days),
//           por lo que este gate no interfiere con la fiabilidad de Koncorde.
// ================================================================
crashBar   = Close < Highest[4](Close[1]) * 0.10
bsCrash    = barssince(crashBar)
hadCrash90 = bsCrash <= crashLookback

// ================================================================
// DETECCIÓN DE PUMP RECIENTE
// ================================================================
maxPump5 = Highest[5](High)

// Nota: en serial offenders con crash reciente, close[5] puede ser el mínimo
// del fondo post-crash → pumpPct aparece muy alto (300%+). Es real: el stock
// colapsó y ahora está en pump de recuperación especulativa. Señal válida.
pumpPct    = (maxPump5 - close[5]) / (close[5] + 0.001) * 100
pumpFuerte = pumpPct >= pumpMinPct

extendido = close > close[5]

distDeMaximo = (maxPump5 - close) / (maxPump5 + 0.001) * 100
noTardio     = distDeMaximo < 40   // [FIX v3.1] de 50% a 40% — serial offenders colapsan, no declinan

volPumpMax = Highest[5](Volume) / (volMedia20 + 0.001)

// ================================================================
// BLOQUE KONCORDE — señales de distribución institucional
// ================================================================
nvi    = NegativeVolumeIndex(close)
nvim   = ExponentialAverage[3](nvi)
nvimax = Highest[90](nvim)
nvimin = Lowest[90](nvim)
azul   = (nvi - nvim) * 100 / Max(nvimax - nvimin, 0.001)

// [K0] Premium: precio sube pero azul ya cae — distribución silenciosa pre-cruce
divergenciaBaj = azul < azul[3] AND close > close[3]
// [K1] Salida activa: azul cayendo 3 sesiones y aún no muy positivo
azulAceleraBaj = azul < azul[1] AND azul[1] < azul[2] AND azul < 10
// [K2] Institucionales ya fuera
azulBajoCero   = azul < 0

hayDistribucion = divergenciaBaj OR azulAceleraBaj OR azulBajoCero

// ================================================================
// BLOQUE VIGÍA10 — giro de momentum
// ================================================================
ignored, ignored, ignored, ignored, ignored, SV, MR, ignored = CALL "Blai5 VIGIA 10"
filtro = ExponentialAverage[20](SV)

svBajando    = SV < SV[1] AND SV[1] < SV[2]
crossSV0neg  = SV CROSSES UNDER 0
bsSV0neg     = barssince(crossSV0neg)
crossSVMRneg = SV CROSSES UNDER MR
bsSVMRneg    = barssince(crossSVMRneg)

// ================================================================
// BLOQUE ICHIMOKU — deterioro estructural
// ================================================================
tenkan     = (Highest[9](High) + Lowest[9](Low)) / 2
kijun      = (Highest[26](High) + Lowest[26](Low)) / 2
ssa0       = (tenkan + kijun) / 2
ssb0       = (Highest[52](High) + Lowest[52](Low)) / 2
ssa        = ssa0[26]
ssb        = ssb0[26]
kumoBottom = Min(ssa, ssb)

// ================================================================
// SCORING CRASH — frescura del historial [NUEVO v3.1]
// bsCrash ya está calculado para el gate — ahora también suma puntos.
// Crash más reciente = patrón más fresco = retail más predispuesto al FOMO.
// ================================================================

// [C1] 20 pts — Crash muy reciente (≤60 días) — el mismo ciclo todavía activo
sC1 = (bsCrash <= 60) * 20

// [C2] 10 pts — Crash menos reciente (60-180 días) — patrón reconocible
sC2 = (bsCrash > 60 AND bsCrash <= 180) * 10

scoreC = sC1 + sC2   // máx teórico: 20 (mutuamente excluyentes)

// ================================================================
// SCORING KONCORDE — distribución institucional
// ================================================================

// [K0] 35 pts — SEÑAL PREMIUM: distribución silenciosa pre-cruce
sK0 = divergenciaBaj * 35

// [K1] 20 pts — Azul cayendo 3 sesiones (azul < 10)
sK1 = azulAceleraBaj * 20

// [K2] 15 pts — Azul bajo cero (institucionales ya fuera)
sK2 = azulBajoCero * 15

// [K3] 10 pts — Azul bajo cero Y acelerando
sK3 = (azulBajoCero AND azulAceleraBaj) * 10

scoreK = sK0 + sK1 + sK2 + sK3   // máx teórico: 80

// ================================================================
// SCORING VIGÍA10 — flip de momentum bajista
// ================================================================

// [V1] 30 pts — SV cruza 0 hacia abajo
sV1 = (bsSV0neg <= 3 AND SV < 0) * 30

// [V2] 20 pts — SV cruza bajo MR (señal temprana)
sV2 = (bsSVMRneg <= 5) * 20

// [V3] 15 pts — SV negativo y cayendo
sV3 = (SV < 0 AND svBajando) * 15

// [V4] 10 pts — Filtro girando a la baja
sV4 = (filtro < filtro[1] AND filtro < filtro[3]) * 10

// [V5] 8 pts — SV < MR y filtro negativo
sV5 = (SV < MR AND filtro < 0) * 8

scoreV = sV1 + sV2 + sV3 + sV4 + sV5   // máx teórico: 83

// ================================================================
// SCORING ICHIMOKU — deterioro estructural de precio
// ================================================================

// [I1] 20 pts — Precio bajo la Kijun
sI1 = (close < kijun) * 20

// [I2] 10 pts — Precio bajo la Tenkan pero sobre Kijun
sI2 = (close < tenkan AND close >= kijun) * 10

// [I3] 8 pts — Nube bajista futura (SSB > SSA)
sI3 = (ssb > ssa) * 8

// [I4] 5 pts — Precio bajo la nube
sI4 = (close < kumoBottom) * 5

scoreI = sI1 + sI2 + sI3 + sI4   // máx teórico: 43

// ================================================================
// SCORING PUMP — calidad y extensión del pump detectado
// ================================================================

// [P1] 15 pts — Pump muy violento (>= 60% en 5 días)
sP1 = (pumpPct >= 60) * 15

// [P2] 8 pts — Pump moderado (30–60%)
sP2 = (pumpPct >= pumpMinPct AND pumpPct < 60) * 8

// [P3] 12 pts — Precio extremadamente extendido sobre media de 20 días
sP3 = (close > Average[20](close) * 1.5) * 12

// [P4] 8 pts — Moderadamente extendido sobre media
sP4 = (close > Average[20](close) * 1.2 AND close <= Average[20](close) * 1.5) * 8

// [P5] 12 pts — Volatilidad extrema (ATR relativo > 10%)
sP5 = (atrRelativo > 0.10) * 12

// [P6] 6 pts — Volatilidad alta (ATR 5–10%)
sP6 = (atrRelativo > 0.05 AND atrRelativo <= 0.10) * 6

// [P7] 10 pts — Inicio de caída desde el techo (10–30% desde el máximo)
sP7 = (distDeMaximo >= 10 AND distDeMaximo < 30) * 10

// [P8] 15 pts — Pump con volumen explosivo (>= 5x media 20 días)
sP8 = (volPumpMax >= 5) * 15

// [P9] 8 pts — Pump con volumen elevado (2x–5x media)
sP9 = (volPumpMax >= 2 AND volPumpMax < 5) * 8

scoreP = sP1 + sP2 + sP3 + sP4 + sP5 + sP6 + sP7 + sP8 + sP9   // máx teórico: 94

// ================================================================
// SCORE FINAL Y CONDICIÓN
// ================================================================
score = scoreC + scoreK + scoreV + scoreI + scoreP   // máx teórico: 320

// JERARQUÍA DE PUERTAS (orden de importancia operativa):
//   hadCrash90:      [PRIMARIA] serial offender confirmado — crash >90% en ≤180 días
//   hayDistribucion: [SECUNDARIA] Koncorde confirma salida institucional activa
//   pumpFuerte:      pump real ≥ pumpMinPct% en últimos 5 días
//   extendido:       precio aún sobre el nivel pre-pump
//   noTardio:        ≤40% desde el pico (serial offenders colapsan rápido)
//   baseOk:          precio, liquidez y volatilidad mínima
//   noEnPump:        no subiendo >15% hoy (pump todavía activo)
condicion = hadCrash90 AND hayDistribucion AND pumpFuerte AND extendido AND noTardio AND baseOk AND noEnPump AND score >= scoreMin

SCREENER[score * condicion](score AS "Score", bsCrash AS "DiasUltCrash", pumpPct AS "Pump%", distDeMaximo AS "DistMax%", movHoy AS "Hoy%", divergenciaBaj AS "DivBaj", azul AS "Azul", SV AS "SV", scoreK AS "ScoreK", scoreV AS "ScoreV", close AS "Precio", dolarVol AS "DolarVol")
