// ============================================================
// KONCORDE PRIMAVERA NACIENTE + ICHIMOKU v2.0
// Captura el INICIO de la Primavera antes de la explosion
// Azul recien cruzado (institucionales entrando) + Verde bajo
// Precio sobre nube con rotura reciente + Vigia10 girando
// Timeframe: DIARIO
// Si salen 0 resultados: sube bsAzulMax a 5 o baja minVolDolar a 500000
// Si salen demasiados:   baja bsAzulMax a 1 o sube minVolDolar a 2000000
// ============================================================

// ---- PARAMETROS AJUSTABLES ----
bsAzulMax  = 3       // barras desde que azul cruzo cero (0=hoy, 1=ayer, 2=anteayer...)
minVolDolar = 1000000 // volumen medio diario en dolares (20 sesiones)
minPrice   = 5
maxPrice   = 500

// ---- KONCORDE ----
m = 3

pvi = PositiveVolumeIndex(close)
pvim = ExponentialAverage[m](pvi)
pvimax = Highest[90](pvim)
pvimin = Lowest[90](pvim)
oscp = (pvi - pvim) * 100 / Max(pvimax - pvimin, 0.001)

nvi = NegativeVolumeIndex(close)
nvim = ExponentialAverage[m](nvi)
nvimax = Highest[90](nvim)
nvimin = Lowest[90](nvim)
azul = (nvi - nvim) * 100 / Max(nvimax - nvimin, 0.001)

xmf = MoneyFlowIndex[14]
OB1 = (BollingerUp[25](TotalPrice) + BollingerDown[25](TotalPrice)) / 2
OB2 = Max(BollingerUp[25](TotalPrice) - BollingerDown[25](TotalPrice), 0.001)
BollOsc = ((TotalPrice - OB1) / OB2) * 100
xrsi = RSI[14](TotalPrice)
STOC = Stochastic[21,3](TotalPrice)
marron = (xrsi + xmf + BollOsc + (STOC / 3)) / 2
verde = marron + oscp
media = ExponentialAverage[m](marron)

// ---- VIGIA10 ----
ignored, ignored, ignored, ignored, ignored, filtro, ignored, ignored = CALL "Blai5 VIGIA 10"

// ---- ICHIMOKU ----
tenkan = (Highest[9](High) + Lowest[9](Low)) / 2
kijun  = (Highest[26](High) + Lowest[26](Low)) / 2
ssa0   = (tenkan + kijun) / 2
ssb0   = (Highest[52](High) + Lowest[52](Low)) / 2
ssa    = ssa0[26]
ssb    = ssb0[26]
kumoTop = Max(ssa, ssb)

// ---- FILTROS BASICOS ----
volDolar = Average[20](Volume * Close)
cBase = volDolar >= minVolDolar AND Close >= minPrice AND Close <= maxPrice

// ---- KONCORDE: primavera naciente ----
// Azul recien cruzado: institucionales acaban de entrar
crossAzul  = azul CROSSES OVER 0
bsAzul     = barssince(crossAzul)
cK1 = bsAzul <= bsAzulMax

// Verde positivo: primavera confirmada (azul recien cruzado = señal temprana)
cK2 = verde > 0

// Marron acelerando por encima de su media
cK3 = marron > media

// Azul creciendo tras el cruce (momentum institucional continua)
cK4 = azul > azul[1]

// ---- ICHIMOKU: tendencia mayor de calidad ----
cI1 = close > kumoTop      // precio sobre nube
cI2 = tenkan > kijun       // tenkan sobre kijun
cI3 = ssa > ssb            // nube alcista (A sobre B)
cI4 = close > Close[26]    // chikou libre (sin resistencia en el pasado)

// ---- VIGIA10: positivo ----
cV1 = filtro > 0

condicion = cBase AND cK1 AND cK2 AND cK3 AND cK4 AND cI1 AND cI2 AND cI3 AND cI4 AND cV1

SCREENER[condicion](bsAzul AS "BsAzul", verde AS "Verde", azul AS "Azul", filtro AS "Filtro", Close AS "Precio")
